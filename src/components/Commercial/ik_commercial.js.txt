frappe.pages['ik-commercial'].on_page_load = function (wrapper) {

    if (!document.getElementById('ik-com-css')) {
        var style = document.createElement('style');
        style.id = 'ik-com-css';
        style.textContent = IK_COM._css();
        document.head.appendChild(style);
    }

    if (!document.getElementById('ik-com-fonts')) {
        var link = document.createElement('link');
        link.id = 'ik-com-fonts';
        link.rel = 'stylesheet';
        link.href = 'https://fonts.googleapis.com/css2?family=Nunito:wght@400;600;700;800;900&family=Jost:wght@400;500;600;700;800&family=JetBrains+Mono:wght@400;600;700&display=swap';
        document.head.appendChild(link);
    }

    frappe.ui.make_app_page({
        parent: wrapper,
        title: 'Commercial',
        single_column: true,
    });

    var $body = $(wrapper).find('.layout-main-section');
    if (!$body.length) $body = $(wrapper).find('.page-content');
    $body.css({ padding: '0' }).html(IK_COM.getHTML());
    $(wrapper).find('.layout-main-section-wrapper').css('padding', '0');

    IK_COM.init();
};

    // ── Stepper ───────────────────────────────────────────────────────────────
    _gotoStep: function (idx) {
        if (idx < 0 || idx >= this._pkgStepSequence.length) return;
        this._pkgStepIdx = idx;
        // Update stepper items
        var self = this;
        $('#ik-pkg-stepper .ik-com-stepper-item').each(function () {
            var i = parseInt($(this).data('step'));
            $(this).removeClass('active done');
            if (i < idx) $(this).addClass('done');
            else if (i === idx) $(this).addClass('active');
        });
        // Show/hide step content panels
        $('#screen-pkg-detail .ik-com-tab-pane[data-pane="execution"] .ik-com-step').hide().removeClass('active');
        $('#screen-pkg-detail .ik-com-tab-pane[data-pane="execution"] .ik-com-step[data-step="' + idx + '"]').show().addClass('active');
        // Update action bar
        $('#ik-step-current').text(idx + 1);
        $('#btn-step-prev').prop('disabled', idx === 0);
        var isLast = (idx === self._pkgStepSequence.length - 1);
        if (isLast) {
            $('#btn-step-next').text('✉ Lock for tender →');
            $('#ik-step-state').text('Ready for tender').attr('class', 'state-ready');
        } else {
            $('#btn-step-next').text('Next →');
            $('#ik-step-state').text('Defining').attr('class', 'state-defining');
        }
    },

    _nextStep: function () {
        if (this._pkgStepIdx < this._pkgStepSequence.length - 1) {
            this._gotoStep(this._pkgStepIdx + 1);
        } else {
            $('#ik-tender-modal').show();
        }
    },

    _prevStep: function () {
        if (this._pkgStepIdx > 0) this._gotoStep(this._pkgStepIdx - 1);
    },
